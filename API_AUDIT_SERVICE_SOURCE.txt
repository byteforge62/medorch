import type { ScheduleStatus, UserRole } from "@/generated/prisma/client";

const ALLOWED_STATUS_TRANSITIONS = {
  SCHEDULED: ["CONFIRMED", "CANCELLED"],
  CONFIRMED: ["IN_PROGRESS", "DELAYED", "CANCELLED"],
  IN_PROGRESS: ["COMPLETED", "DELAYED"],
  DELAYED: ["CONFIRMED", "IN_PROGRESS", "CANCELLED"],
  COMPLETED: [],
  CANCELLED: [],
} satisfies Record<ScheduleStatus, readonly ScheduleStatus[]>;

export function isScheduleStatusTransitionAllowed(
  currentStatus: ScheduleStatus,
  nextStatus: ScheduleStatus,
) {
  return ALLOWED_STATUS_TRANSITIONS[currentStatus].some(
    (status) => status === nextStatus,
  );
}

export function canReadIndividualSchedule(
  role: UserRole,
  isAssignedToUser: boolean,
) {
  return role !== "OT_STAFF" || isAssignedToUser;
}

export function redactScheduleClinicalNotes<
  T extends { clinicalNotes: unknown },
>(schedule: T) {
  return {
    ...schedule,
    clinicalNotes: null,
  };
}

export function redactScheduleDetailsForOT<
  T extends {
    clinicalNotes: unknown;
    notes: unknown;
    patient: {
      email: unknown;
      dateOfBirth: unknown;
      gender: unknown;
    };
  },
>(schedule: T) {
  return {
    ...schedule,
    clinicalNotes: null,
    notes: [],
    patient: {
      ...schedule.patient,
      email: null,
      dateOfBirth: null,
      gender: null,
    },
  };
}

export async function createScheduleWithValidation(
  data: {
    patientId: string;
    departmentId: string;
    otRoomId: string;
    surgeonId: string;
    createdById: string;
    procedure: string;
    scheduledDate: Date;
    startTime: Date;
    endTime: Date;
    priority?:
    | "ELECTIVE"
    | "URGENT"
    | "EMERGENCY";
    clinicalNotes?: string;
  },
  actorUserId?: string
) {
  const patient = await findPatientById(data.patientId);

  if (!patient) {
    throw new Error("PATIENT_NOT_FOUND");
  }

  const department = await findDepartmentById(
    data.departmentId,
  );

  if (!department) {
    throw new Error("DEPARTMENT_NOT_FOUND");
  }

  if (!department.isActive) {
    throw new Error("DEPARTMENT_INACTIVE");
  }

  const otRoom = await findOTRoomForSchedule(
    data.otRoomId,
  );

  if (!otRoom) {
    throw new Error("OT_ROOM_NOT_FOUND");
  }

  if (!otRoom.isActive) {
    throw new Error("OT_ROOM_INACTIVE");
  }

  if (otRoom.departmentId !== data.departmentId) {
    throw new Error("OT_ROOM_DEPARTMENT_MISMATCH");
  }

  if (otRoom.status === "MAINTENANCE") {
    throw new Error("OT_ROOM_MAINTENANCE");
  }

  if (otRoom.status === "DISABLED") {
    throw new Error("OT_ROOM_DISABLED");
  }

  const surgeon = await findSurgeonById(
    data.surgeonId,
  );

  if (!surgeon) {
    throw new Error("SURGEON_NOT_FOUND");
  }

  if (surgeon.role !== "DOCTOR") {
    throw new Error("INVALID_SURGEON");
  }

  if (surgeon.status !== "ACTIVE") {
    throw new Error("SURGEON_INACTIVE");
  }

  if (data.endTime <= data.startTime) {
    throw new Error("INVALID_TIME_RANGE");
  }

  const conflicts = await findScheduleConflicts({
    patientId: data.patientId,
    otRoomId: data.otRoomId,
    surgeonId: data.surgeonId,
    startTime: data.startTime,
    endTime: data.endTime,
  });

  if (conflicts.length > 0) {
    const roomConflict = conflicts.some(
      (conflict) => conflict.otRoomId === data.otRoomId,
    );

    if (roomConflict) {
      throw new Error("OT_ROOM_SCHEDULE_CONFLICT");
    }

    const surgeonConflict = conflicts.some(
      (conflict) => conflict.surgeonId === data.surgeonId,
    );

    if (surgeonConflict) {
      throw new Error("SURGEON_SCHEDULE_CONFLICT");
    }

    const patientConflict = conflicts.some(
      (conflict) => conflict.patientId === data.patientId,
    );

    if (patientConflict) {
      throw new Error("PATIENT_SCHEDULE_CONFLICT");
    }
  }
  const schedule = await createScheduleRecord(data);

  await recordAudit({
    userId: actorUserId,
    action: "CREATE",
    entity: "Schedule",
    entityId: schedule.id,
    description: `Schedule for procedure "${schedule.procedure}" was created.`,
    metadata: {
      patientId: schedule.patientId,
      departmentId: schedule.departmentId,
      otRoomId: schedule.otRoomId,
      surgeonId: schedule.surgeonId,
      priority: schedule.priority,
    },
  });

  return schedule;
}

export async function updateScheduleWithValidation(
  id: string,
  data: {
    patientId?: string;
    departmentId?: string;
    otRoomId?: string;
    surgeonId?: string;
    procedure?: string;
    scheduledDate?: Date;
    startTime?: Date;
    endTime?: Date;
    priority?:
    | "ELECTIVE"
    | "URGENT"
    | "EMERGENCY";
    clinicalNotes?: string | null;
  },
  actorUserId?: string
) {
  const existingSchedule = await findScheduleById(id);

  if (!existingSchedule) {
    throw new Error("SCHEDULE_NOT_FOUND");
  }

  if (
    existingSchedule.status === "COMPLETED" ||
    existingSchedule.status === "CANCELLED"
  ) {
    throw new Error("SCHEDULE_IMMUTABLE");
  }

  const patientId =
    data.patientId ?? existingSchedule.patientId;

  const departmentId =
    data.departmentId ?? existingSchedule.departmentId;

  const otRoomId =
    data.otRoomId ?? existingSchedule.otRoomId;

  const surgeonId =
    data.surgeonId ?? existingSchedule.surgeonId;

  const startTime =
    data.startTime ?? existingSchedule.startTime;

  const endTime =
    data.endTime ?? existingSchedule.endTime;

  const patient = await findPatientById(patientId);

  if (!patient) {
    throw new Error("PATIENT_NOT_FOUND");
  }

  const department = await findDepartmentById(departmentId);

  if (!department) {
    throw new Error("DEPARTMENT_NOT_FOUND");
  }

  if (!department.isActive) {
    throw new Error("DEPARTMENT_INACTIVE");
  }

  const otRoom = await findOTRoomForSchedule(otRoomId);

  if (!otRoom) {
    throw new Error("OT_ROOM_NOT_FOUND");
  }

  if (!otRoom.isActive) {
    throw new Error("OT_ROOM_INACTIVE");
  }

  if (otRoom.departmentId !== departmentId) {
    throw new Error("OT_ROOM_DEPARTMENT_MISMATCH");
  }

  if (
    otRoom.status === "MAINTENANCE" ||
    otRoom.status === "DISABLED"
  ) {
    throw new Error("OT_ROOM_UNAVAILABLE");
  }

  const surgeon = await findSurgeonById(surgeonId);

  if (!surgeon) {
    throw new Error("SURGEON_NOT_FOUND");
  }

  if (surgeon.role !== "DOCTOR") {
    throw new Error("INVALID_SURGEON");
  }

  if (surgeon.status !== "ACTIVE") {
    throw new Error("SURGEON_INACTIVE");
  }

  if (endTime <= startTime) {
    throw new Error("INVALID_TIME_RANGE");
  }

  const conflicts = await findScheduleConflicts({
    patientId,
    otRoomId,
    surgeonId,
    startTime,
    endTime,
    excludeScheduleId: id,
  });

  if (conflicts.length > 0) {
    const roomConflict = conflicts.some(
      (conflict) => conflict.otRoomId === otRoomId,
    );

    if (roomConflict) {
      throw new Error("OT_ROOM_SCHEDULE_CONFLICT");
    }

    const surgeonConflict = conflicts.some(
      (conflict) => conflict.surgeonId === surgeonId,
    );

    if (surgeonConflict) {
      throw new Error("SURGEON_SCHEDULE_CONFLICT");
    }
  }

  const patientConflict = conflicts.some(
    (conflict) => conflict.patientId === patientId,
  );

  if (patientConflict) {
    throw new Error("PATIENT_SCHEDULE_CONFLICT");
  }

  return updateScheduleRecord(id, data, actorUserId);
}

export async function assignScheduleStaffById(
  scheduleId: string,
  data: {
    userId: string;
    role:
    | "SURGEON"
    | "NURSE"
    | "ANESTHETIST"
    | "TECHNICIAN"
    | "OTHER";
  },
  actorUserId?: string
) {
  const schedule = await findScheduleById(scheduleId);

  if (!schedule) {
    throw new Error("SCHEDULE_NOT_FOUND");
  }

  const user = await findUserForStaffAssignment(
    data.userId,
  );

  if (!user) {
    throw new Error("STAFF_USER_NOT_FOUND");
  }

  if (user.status !== "ACTIVE") {
    throw new Error("STAFF_USER_INACTIVE");
  }

  const doctorRoles = ["SURGEON", "ANESTHETIST"] as const;

  if (doctorRoles.includes(data.role as (typeof doctorRoles)[number])) {
    if (user.role !== "DOCTOR" || !user.doctorProfile) {
      throw new Error("STAFF_ROLE_MISMATCH");
    }
  } else if (user.role !== "OT_STAFF") {
    throw new Error("STAFF_ROLE_MISMATCH");
  }

  // The repository creates the assignment and its audit record
  // within the same transaction.
  return createScheduleStaff(
    {
      scheduleId,
      userId: data.userId,
      role: data.role,
    },
    actorUserId,
  );
}

export async function updateScheduleStatus(
  id: string,
  status:
    | "SCHEDULED"
    | "CONFIRMED"
    | "IN_PROGRESS"
    | "COMPLETED"
    | "CANCELLED"
    | "DELAYED",
  expectedStatus:
    | "SCHEDULED"
    | "CONFIRMED"
    | "IN_PROGRESS"
    | "COMPLETED"
    | "CANCELLED"
    | "DELAYED",
  actorUserId?: string
) {
  return prisma.$transaction(async (tx) => {
    const update = await tx.schedule.updateMany({
      where: {
        id,
        status: expectedStatus,
      },
      data: {
        status,
      },
    });

    if (update.count !== 1) {
      const schedule = await tx.schedule.findUnique({
        where: { id },
        select: { id: true },
      });

      throw new Error(
        schedule ? "SCHEDULE_STATUS_CONFLICT" : "SCHEDULE_NOT_FOUND",
      );
    }

    const schedule = await tx.schedule.findUniqueOrThrow({
      where: { id },
      select: {
        id: true,
        procedure: true,
        scheduledDate: true,
        startTime: true,
        endTime: true,
        status: true,
        priority: true,
        updatedAt: true,
      },
    });

    const auditAction =
      status === "COMPLETED"
        ? "COMPLETE"
        : status === "CANCELLED"
          ? "CANCEL"
          : "UPDATE";

    await createAuditLog(
      {
        userId: actorUserId,
        action: auditAction,
        entity: "Schedule",
        entityId: schedule.id,
        description: `Schedule status changed to ${status}.`,
        metadata: { status },
      },
      tx,
    );

    return schedule;
  });
}

export async function createScheduleEquipment(
  data: { scheduleId: string; equipmentId: string },
  actorUserId?: string,
) {
  return prisma.$transaction(async (tx) => {
    // Claim the equipment only if it is still available.
    const claim = await tx.equipment.updateMany({
      where: { id: data.equipmentId, status: "AVAILABLE" },
      data: { status: "IN_USE" },
    });

    if (claim.count !== 1) {
      const equipment = await tx.equipment.findUnique({
        where: { id: data.equipmentId },
        select: { id: true },
      });
      if (!equipment) throw new Error("EQUIPMENT_NOT_FOUND");
      throw new Error("EQUIPMENT_UNAVAILABLE");
    }

    const assignment = await tx.scheduleEquipment.create({
      data,
      select: {
        id: true,
        scheduleId: true,
        equipmentId: true,
        assignedAt: true,
        releasedAt: true,
        equipment: {
          select: {
            id: true,
            name: true,
            category: true,
            serialNumber: true,
            status: true,
          },
        },
      },
    });

    await createAuditLog({
      userId: actorUserId,
      action: "CREATE",
      entity: "ScheduleEquipment",
      entityId: assignment.id,
      description: "Equipment assigned to schedule.",
      metadata: { scheduleId: assignment.scheduleId, equipmentId: assignment.equipmentId },
    }, tx);

    return assignment;
  });
}


export async function releaseScheduleEquipment(
  id: string,
  scheduleId: string,
  actorUserId?: string,
) {
  return prisma.$transaction(async (tx) => {
    const assignment = await tx.scheduleEquipment.findUnique({
      where: { id },
      select: {
        id: true,
        scheduleId: true,
        equipmentId: true,
        assignedAt: true,
        releasedAt: true,
      },
    });

    if (!assignment || assignment.scheduleId !== scheduleId) {
      throw new Error("EQUIPMENT_ASSIGNMENT_NOT_FOUND");
    }

    if (assignment.releasedAt !== null) {
      throw new Error("EQUIPMENT_ALREADY_RELEASED");
    }

    const released = await tx.scheduleEquipment.updateMany({
      where: {
        id,
        scheduleId,
        releasedAt: null,
      },
      data: {
        releasedAt: new Date(),
      },
    });

    if (released.count !== 1) {
      throw new Error("EQUIPMENT_ALREADY_RELEASED");
    }

    const activeAssignments = await tx.scheduleEquipment.count({
      where: {
        equipmentId: assignment.equipmentId,
        releasedAt: null,
      },
    });

    if (activeAssignments === 0) {
      await tx.equipment.updateMany({
        where: {
          id: assignment.equipmentId,
          status: "IN_USE",
        },
        data: {
          status: "AVAILABLE",
        },
      });
    }

    const result = {
      ...assignment,
      releasedAt: new Date(),
    };

    await createAuditLog({
      userId: actorUserId,
      action: "UPDATE",
      entity: "ScheduleEquipment",
      entityId: assignment.id,
      description: "Equipment released from schedule.",
      metadata: { scheduleId: assignment.scheduleId, equipmentId: assignment.equipmentId },
    }, tx);

    return result;
  });
}

export async function resendPatientVerificationEmail(
  email: string,
  ipAddress: string,
) {
  const normalizedEmail = email.trim().toLowerCase();

  if (!normalizedEmail || normalizedEmail.length > 254) {
    return { accepted: true, rateLimited: false };
  }

  const emailHash = createHash("sha256")
    .update(normalizedEmail)
    .digest("hex");

  const ipHash = createHash("sha256")
    .update(ipAddress || "unknown")
    .digest("hex");

  const allowed = await recordVerificationEmailRequest(
    emailHash,
    ipHash,
    "EMAIL_VERIFICATION",
  );

  if (!allowed) {
    return {
      accepted: true,
      rateLimited: true,
    };
  }

  const token = generateEmailVerificationToken();
  const tokenHash = hashEmailVerificationToken(token);
  const expiresAt = getEmailVerificationExpiry();

  const user = await createPatientResendVerificationToken(
    normalizedEmail,
    tokenHash,
    expiresAt,
  );

  if (user) {
    try {
      await sendVerificationEmail(user.email, user.name, token);
    } catch (error) {
      console.error("Resend verification email failed:", error);
    }
  }

  return { accepted: true, rateLimited: false };
}

export async function verifyPatientEmail(tokenHash: string): Promise<boolean> {
  return prisma.$transaction(async (tx) => {
    const verificationToken = await tx.emailVerificationToken.findUnique({
      where: { tokenHash },
      select: {
        userId: true,
        expiresAt: true,
        user: {
          select: {
            emailVerifiedAt: true,
          },
        },
      },
    });

    if (!verificationToken) {
      return false;
    }

    if (verificationToken.expiresAt <= new Date()) {
      await tx.emailVerificationToken.deleteMany({
        where: { tokenHash },
      });

      return false;
    }

    const consumedToken = await tx.emailVerificationToken.deleteMany({
      where: {
        tokenHash,
        expiresAt: { gt: new Date() },
      },
    });

    if (consumedToken.count === 0) {
      return false;
    }

    if (!verificationToken.user.emailVerifiedAt) {
      await tx.user.update({
        where: { id: verificationToken.userId },
        data: { emailVerifiedAt: new Date() },
      });
    }

    // Invalidate all outstanding verification tokens for this user.
    await tx.emailVerificationToken.deleteMany({
      where: { userId: verificationToken.userId },
    });

    return true;
  });
}

type VerificationRequestPurpose =
  | "EMAIL_VERIFICATION"
  | "PASSWORD_RESET"
  | "PASSWORD_RESET_ATTEMPT"
  | "LOGIN_ATTEMPT"
  | "REGISTRATION"
  | "EMAIL_VERIFICATION_TOKEN"


export async function isLoginAttemptAllowed(
  emailHash: string,
  ipHash: string,
): Promise<boolean> {
  const windowStart = new Date(Date.now() - 15 * 60 * 1000);

  return prisma.$transaction(async (tx) => {
    const lockKeys = [
      `LOGIN_ATTEMPT:email:${emailHash}`,
      `LOGIN_ATTEMPT:ip:${ipHash}`,
    ].sort();

    for (const lockKey of lockKeys) {
      await tx.$queryRaw`
        SELECT 1 AS locked
        FROM (
          SELECT pg_advisory_xact_lock(hashtext(${lockKey})::bigint)
        ) AS lock_query
      `;
    }

    const [emailAttempts, ipAttempts] = await Promise.all([
      tx.verificationEmailRequest.count({
        where: {
          purpose: "LOGIN_ATTEMPT",
          emailHash,
          createdAt: { gte: windowStart },
        },
      }),
      tx.verificationEmailRequest.count({
        where: {
          purpose: "LOGIN_ATTEMPT",
          ipHash,
          createdAt: { gte: windowStart },
        },
      }),
    ]);

    return emailAttempts < 5 && ipAttempts < 20;
  });
}

export async function recordVerificationEmailRequest(
  emailHash: string,
  ipHash: string,
  purpose: VerificationRequestPurpose = "EMAIL_VERIFICATION"
): Promise<boolean> {
  const now = new Date();
  const windowStart = new Date(now.getTime() - 15 * 60 * 1000);
  const cleanupBefore = new Date(now.getTime() - 24 * 60 * 60 * 1000);

  return prisma.$transaction(async (tx) => {
    const lockKeys = [`${purpose}:email:${emailHash}`, `${purpose}:ip:${ipHash}`].sort();

    for (const lockKey of lockKeys) {
      await tx.$queryRaw`
    SELECT 1 AS locked
    FROM (
      SELECT pg_advisory_xact_lock(hashtext(${lockKey})::bigint)
    ) AS lock_query
  `;
    }

    await tx.verificationEmailRequest.deleteMany({
      where: { createdAt: { lt: cleanupBefore } },
    });

    const [emailRequests, ipRequests] = await Promise.all([
      tx.verificationEmailRequest.count({
        where: {
          purpose,
          emailHash,
          createdAt: { gte: windowStart },
        },
      }),
      tx.verificationEmailRequest.count({
        where: {
          purpose,
          ipHash,
          createdAt: { gte: windowStart },
        },
      }),
    ]);

    const emailLimit = purpose === "LOGIN_ATTEMPT" ? 5 : 3;
    const ipLimit = purpose === "LOGIN_ATTEMPT" ? 20 : 10;

    if (emailRequests >= emailLimit || ipRequests >= ipLimit) {
      return false;
    }

    await tx.verificationEmailRequest.create({
      data: { purpose, emailHash, ipHash },
    });

    return true;
  });
}

export async function resetPasswordWithToken(
  tokenHash: string,
  passwordHash: string,
): Promise<boolean> {
  return prisma.$transaction(async (tx) => {
    const resetTokens = await tx.$queryRaw<
      Array<{
        id: string;
        userId: string;
        expiresAt: Date;
      }>
    >`
      SELECT "id", "userId", "expiresAt"
      FROM "PasswordResetToken"
      WHERE "tokenHash" = ${tokenHash}
      FOR UPDATE
    `;

    const resetToken = resetTokens[0];

    if (!resetToken) {
      return false;
    }

    if (resetToken.expiresAt <= new Date()) {
      await tx.passwordResetToken.delete({
        where: { id: resetToken.id },
      });

      return false;
    }

    await tx.user.update({
      where: { id: resetToken.userId },
      data: {
        passwordHash,
        sessionVersion: {
          increment: 1,
        },
      },
    });

    await tx.passwordResetToken.deleteMany({
      where: { userId: resetToken.userId },
    });

    return true;
  });
}

export interface FindAuditLogsOptions {
  userId?: string;
  action?: AuditAction;
  entity?: string;
  entityId?: string;
  fromDate?: Date;
  toDate?:Date;
  limit?: number;
  offset?: number;

function buildAuditLogWhere(
  options: FindAuditLogsOptions = {},
): Prisma.AuditLogWhereInput {
  const {
    userId,
    action,
    entity,
    entityId,
    fromDate,
    toDate,
  } = options;

  return {
    ...(userId ? { userId } : {}),
    ...(action ? { action } : {}),
    ...(entity ? { entity } : {}),
    ...(entityId ? { entityId } : {}),
    ...(fromDate || toDate
      ? {
          createdAt: {
            ...(fromDate ? { gte: fromDate } : {}),
            ...(toDate ? { lte: toDate } : {}),
          },
        }
      : {}),
  };
}
