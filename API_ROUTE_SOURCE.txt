import { markAllAlertsAsRead } from "@/modules/alerts/alerts.service";
import { z } from "zod";

const readAllSchema = z.object({
  userId: z.string().uuid().optional(),
});

export async function PATCH(request: Request) {
  const requestId = getRequestId(request);

  try {
    const { session, response } = await authorizeApiPermission(
      ["alerts:update","alerts:update:own"],
      requestId,
    );

    if (!session) {
      return response;
    }

    let targetUserId = session.user.id;
    const rawBody = await request.text();

    if (rawBody.trim()) {
      let body: unknown;

      try {
        body = JSON.parse(rawBody);
      } catch {
        return apiError(
          "Invalid JSON request body.",
          400,
          undefined,
          requestId,
        );
      }

      const validation = readAllSchema.safeParse(body);

      if (!validation.success) {
        return apiError(
          "Invalid request body.",
          400,
          validation.error.flatten(),
          requestId,
        );
      }

      if (validation.data.userId) {
        targetUserId = validation.data.userId;
      }
    }

    const result = await markAllAlertsAsRead(targetUserId, session.user);

    return apiSuccess(result, 200);
  } catch (error) {
    console.error(`[${requestId}] PATCH /api/alerts/read-all error:`, error);

    if (error instanceof Error && error.message === "FORBIDDEN_ALERT_ACCESS") {
      return apiError(
        "You do not have permission to mark alerts as read for this user.",

    }

    return apiError("Failed to mark alerts as read.", 500, undefined, requestId);

      return response;

      return apiError(

import { z } from "zod";
import { authorizeApiPermission } from "@/lib/api/auth";
import { getRequestId } from "@/lib/api/request-id";
import { apiError, apiSuccess } from "@/lib/api/response";
import { getAuditLogs, getAuditLogCount } from "@/modules/audit/audit.service";
import type { AuditAction } from "@/generated/prisma/client";

const auditQuerySchema = z.object({
  userId: z.string().uuid().optional(),
  action: z
    .enum([
      "CREATE",
      "UPDATE",
      "DELETE",
      "LOGIN",
      "LOGOUT",
      "APPROVE",
      "REJECT",
      "CANCEL",
      "COMPLETE",
    ])
    .optional(),
  entity: z.string().trim().min(1).max(100).optional(),
  entityId: z.string().trim().min(1).max(100).optional(),
  fromDate: z.iso.date().optional(),
  toDate: z.iso.date().optional(),
  limit: z.coerce.number().int().min(1).max(100).default(50),
  offset: z.coerce.number().int().min(0).default(0),
}).refine(
  (data) => 
    !data.fromDate ||
  !data.toDate || 
  data.fromDate <= data.toDate,
  {
    message: "fromDate must be on or before toDate",
    path:["toDate"]
  }
);

export async function GET(request: Request) {
  const requestId = getRequestId(request);

  const { response } = await authorizeApiPermission(
    "audit:read",
    requestId,
  );

  if (response) {
    return response;
  }

  const url = new URL(request.url);

  const parsed = auditQuerySchema.safeParse({
    userId: url.searchParams.get("userId") ?? undefined,
    action: url.searchParams.get("action") ?? undefined,
    entity: url.searchParams.get("entity") ?? undefined,
    entityId: url.searchParams.get("entityId") ?? undefined,
    fromDate: url.searchParams.get("fromDate") ?? undefined,
    toDate: url.searchParams.get("toDate") ?? undefined,
    limit: url.searchParams.get("limit") ?? undefined,
    offset: url.searchParams.get("offset") ?? undefined,
  });

  if (!parsed.success) {
    return apiError(
      "Invalid audit query parameters.",
      400,
      parsed.error.flatten(),
      requestId,
    );
  }

  try {
    const options = {
  ...parsed.data,
  fromDate: parsed.data.fromDate
    ? new Date(`${parsed.data.fromDate}T00:00:00.000Z`)
    : undefined,
  toDate: parsed.data.toDate
    ? new Date(`${parsed.data.toDate}T23:59:59.999Z`)
    : undefined,
  action: parsed.data.action as AuditAction | undefined,
};

const [auditLogs, total] = await Promise.all([
  getAuditLogs(options),
  getAuditLogCount(options),
]);

return apiSuccess(
  {
    items: auditLogs,
    pagination: {
      limit: parsed.data.limit,
      offset: parsed.data.offset,
      total,
      hasMore: parsed.data.offset + auditLogs.length < total,
    },
  },
  200,
);
  } catch (error) {
    console.error(
      `[${requestId}] Failed to fetch audit logs:`,
      error,
    );

    return apiError(
      "Failed to fetch audit logs.",
      500,
      undefined,
      requestId,
    );
  }
}

import { NextRequest } from "next/server";
import { z } from "zod";

import { apiError, apiSuccess } from "@/lib/api/response";
import { readJsonBody } from "@/lib/api/validation";
import { requestPasswordReset } from "@/modules/users/user.service";

const forgotPasswordSchema = z.object({
  email: z.string().trim().email().max(254).transform((value) => value.toLowerCase()),
});

export async function POST(request: NextRequest) {
  const requestId = crypto.randomUUID();

  try {
    const bodyResult = await readJsonBody(request);
    if (!bodyResult.success) {
      return apiError("Invalid JSON request body.", 400, undefined, requestId);
    }

    const parsed = forgotPasswordSchema.safeParse(bodyResult.data);

    if (!parsed.success) {
      return apiError("Please enter a valid email address.", 400, undefined, requestId);
    }

    const forwardedFor = request.headers.get("x-forwarded-for");
    const ipAddress =
      forwardedFor?.split(",")[0]?.trim() ||
      request.headers.get("x-real-ip") ||
      "unknown";

    const result = await requestPasswordReset(
      parsed.data.email,
      ipAddress,
    );

    if (result.rateLimited) {
      return apiSuccess(
        {
          message:
            "You've reached the password reset request limit. Please wait 15 minutes before trying again.",
          rateLimited: true,
        },
        429,
      );
    }

    return apiSuccess(
      {
        message:
          "If an eligible account exists, a password reset email will be sent.",
      },
      202,
    );
  } catch (error) {
    console.error(`[${requestId}] Forgot-password request failed:`, error);

    return apiError(
      "Unable to process your request. Please try again later.",
      500,
      undefined,
      requestId,
    );
  }
}

import { NextResponse } from "next/server";
import { patientRegistrationSchema } from "@/modules/users/user.validation";
import { registerPatient } from "@/modules/users/user.service";
import { createHash } from "node:crypto";
import { recordVerificationEmailRequest } from "@/modules/users/user.repository";

const REGISTRATION_MESSAGE =
  "If registration can be completed, a verification email will be sent to the provided address.";

function isDuplicateEmailError(error: unknown): boolean {
  if (typeof error !== "object" || error === null || !("code" in error)) {
    return false;
  }

  return error.code === "P2002";
}

export async function POST(request: Request) {
  let body: unknown;

  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { message: "Invalid JSON request body." },
      { status: 400 }
    );
  }

  const validation = patientRegistrationSchema.safeParse(body);

  if (!validation.success) {
    return NextResponse.json(
      {
        message: "Validation failed.",
        errors: validation.error.issues.map((issue) => ({
          field: issue.path.join("."),
          message: issue.message,
        })),
      },
      { status: 400 }
    );
  }

  const email = validation.data.email.trim().toLowerCase();

  const forwardedFor = request.headers.get("x-forwarded-for");
  const realIp = request.headers.get("x-real-ip")?.trim();

  const clientIp = forwardedFor?.split(",")[0]?.trim() || realIp;

  const emailHash = createHash("sha256")
    .update(email)
    .digest("hex");

  const ipHash = createHash("sha256")
    .update(clientIp || `unavailable:${email}`)
    .digest("hex");

  const allowed = await recordVerificationEmailRequest(
    emailHash,
    ipHash,
    "REGISTRATION",
  );

  if (!allowed) {
    return NextResponse.json(
      { message: "Too many registration attempts. Please try again later." },
      {
        status: 429,
        headers: { "Retry-After": "900" },
      },
    );
  }

  try {
    await registerPatient(validation.data);

    return NextResponse.json(
      { message: REGISTRATION_MESSAGE },
      { status: 202 }
    );
  } catch (error) {
    if (isDuplicateEmailError(error)) {
      // Avoid disclosing whether an email is already registered.
      return NextResponse.json(
        { message: REGISTRATION_MESSAGE },
        { status: 202 }
      );
    }

    console.error("Patient registration failed:", error);

    return NextResponse.json(
      { message: "Registration could not be completed. Please try again later." },
      { status: 500 }
    );
  }
}

import { completePasswordReset } from "@/modules/users/user.service";

const resetPasswordSchema = z.object({
  token: z.string().regex(/^[a-f0-9]{64}$/i),
  password: z.string().min(12).max(128),
});

export async function POST(request: NextRequest) {
  const requestId = crypto.randomUUID();

  try {
    const bodyResult = await readJsonBody(request);
    if (!bodyResult.success) {
      return apiError("Invalid JSON request body.", 400, undefined, requestId);
    }

    const parsed = resetPasswordSchema.safeParse(bodyResult.data);

    if (!parsed.success) {
      return apiError(
        "Invalid reset link or password. Use a password between 12 and 128 characters.",
        400,
        undefined,
        requestId,
      );
    }

    const forwardedFor = request.headers.get("x-forwarded-for");
    const ipAddress =
      forwardedFor?.split(",")[0]?.trim() ||
      request.headers.get("x-real-ip") ||
      "unknown";

    const ipHash = createHash("sha256")
      .update(ipAddress)
      .digest("hex");

    const tokenHash = createHash("sha256")
      .update(parsed.data.token)
      .digest("hex");

    const allowed = await recordVerificationEmailRequest(
      tokenHash,
      ipHash,

      return apiError(

        429,
        undefined,

    const result = await completePasswordReset(

    );

    if (!result.reset) {
      return apiError(
        "This password reset link is invalid, expired, or already used.",

        undefined,

      );

    return apiSuccess(
      { message: "Your password has been reset successfully." },

    return apiError(
      "Unable to reset your password. Please try again later.",
      500,
      undefined,
      requestId,
    );
  }
}

  const clientIp = forwardedFor?.split(",")[0]?.trim() || realIp || "unknown";

  const tokenHash = createHash("sha256")
    .update(token)
    .digest("hex");

  const ipHash = createHash("sha256")
    .update(clientIp)
    .digest("hex");

  const allowed = await recordVerificationEmailRequest(
    tokenHash,
    ipHash,
    "EMAIL_VERIFICATION_TOKEN",
  );

  if (!allowed) {
    return NextResponse.json(
      { message: "Too many verification attempts. Please try again later." },
      {
        status: 429,
        headers: { "Retry-After": "900" },
      },
    );
  }

      requestId,
    );

    if (!session) {
      return response;
    }

    let body: unknown;

    try {
      body = await request.json();
    } catch {
      return apiError(
        "Invalid JSON request body.",

    const validation = createScheduleSchema.safeParse(body);

    if (!validation.success) {
      return apiError(

      );
    }

    const schedule = await createScheduleWithValidation({
      ...validation.data,
      createdById: session.user.id,
    },

    console.error(
      `[${requestId}] POST /api/schedules error:`,

    if (conflictMessage) {
      return apiError(
        conflictMessage,
        409,
        undefined,
        requestId,
      );
    }

    if (error instanceof Error) {
      const messages: Record<string, string> = {
        PATIENT_NOT_FOUND: "Patient not found.",
        DEPARTMENT_NOT_FOUND: "Department not found.",
        DEPARTMENT_INACTIVE: "Department is inactive.",
        OT_ROOM_NOT_FOUND: "OT room not found.",
        OT_ROOM_INACTIVE: "OT room is inactive.",
        OT_ROOM_DEPARTMENT_MISMATCH:
          "OT room does not belong to the selected department.",
        OT_ROOM_MAINTENANCE:
          "OT room is currently under maintenance.",
        OT_ROOM_DISABLED:
          "OT room is disabled.",
        SURGEON_NOT_FOUND: "Surgeon not found.",
        INVALID_SURGEON:
          "Selected user is not a doctor.",
        SURGEON_INACTIVE:
          "Selected surgeon is not active.",
        OT_ROOM_SCHEDULE_CONFLICT:
          "The selected OT room is already scheduled during this time.",
        SURGEON_SCHEDULE_CONFLICT:
          "The selected surgeon is already scheduled during this time.",
        INVALID_TIME_RANGE:
          "End time must be later than start time.",
      };

      const message = messages[error.message];

      if (message) {
        return apiError(
          message,
          [
            "OT_ROOM_SCHEDULE_CONFLICT",
            "SURGEON_SCHEDULE_CONFLICT",
          ].includes(error.message)
            ? 409
            : 400,
          undefined,
          requestId,
        );
      }
    }

    if (
      error instanceof Error &&
      error.message === "PATIENT_SCHEDULE_CONFLICT"
    ) {
      return apiError(
        "This patient already has a procedure scheduled during the selected time.",
        409,
        undefined,
        requestId,
      );
    }

    return apiError(
      "Failed to create schedule.",
      500,
      undefined,
      requestId,
    );
  }
}

import { userIdSchema } from "@/modules/users/user.validation";

type RouteContext = {
  params: Promise<{ id: string }>;
};

export async function POST(
  request: Request,
  context: RouteContext,
) {
  const requestId = getRequestId(request);

  try {
    const { session, response } = await authorizeApiPermission(
      "users:manage",
      requestId,
    );

    if (!session) {
      return response;
    }

    const { id } = await context.params;

    const idValidation = userIdSchema.safeParse({ id });

    if (!idValidation.success) {
      return apiError(
        "Invalid user ID.",
        400,
        idValidation.error.flatten(),
        requestId,
      );
    }

    const patient = await approvePatient(
      idValidation.data.id,
      session.user.id,
    );

    return apiSuccess(patient, 200);
  } catch (error) {
    if (error instanceof Error) {
      switch (error.message) {
        case "USER_NOT_FOUND":
          return apiError("User not found.", 404, undefined, requestId);
        case "USER_NOT_PATIENT":
          return apiError("Only patient accounts can be approved.", 409, undefined, requestId);
        case "USER_NOT_PENDING":
          return apiError("This account is not pending approval.", 409, undefined, requestId);
        case "EMAIL_NOT_VERIFIED":
          return apiError("The patient must verify their email before approval.", 409, undefined, requestId);
      }
    }

    console.error(`[${requestId}] POST /api/users/${"approve"} error:`, error);

    return apiError(
      "Failed to approve patient.",
      500,
      undefined,
      requestId,
    );
  }
}
